local function getRootPart()
    local ch = player.Character
    return ch and ch:FindFirstChild("HumanoidRootPart")
end

local function getBasePos()
    if cachedBasePos then return cachedBasePos end
    local map = workspace:FindFirstChild("Map")
    local plots = map and map:FindFirstChild("PlotModels")
    local plot = plots and plots:FindFirstChild(player.Name)
    if not plot then return nil end
    local refPos
    for _, part in ipairs(CollectionService:GetTagged("ConveyorCollectPart")) do
        if part:IsDescendantOf(plot) then refPos = part.Position; break end
    end
    if not refPos then
        for _, desc in ipairs(plot:GetDescendants()) do
            if desc:IsA("BasePart") and (desc.Name:find("Collect") or desc.Name:find("Conveyor")) then
                refPos = desc.Position; break
            end
        end
    end
    if refPos then
        local center = plot:GetPivot().Position
        local dir = Vector3.new(center.X - refPos.X, 0, center.Z - refPos.Z)
        if dir.Magnitude > 0.5 then
            local finalPos = refPos - dir.Unit * 16 + Vector3.new(0, 2, 0)
            local lookAt = finalPos - (refPos - finalPos)
            cachedBasePos = CFrame.new(finalPos, lookAt)
            return cachedBasePos
        end
    end
    cachedBasePos = plot:GetPivot().Position
    return cachedBasePos
end

local function teleportTo(target)
    if not target then return end
    local root = getRootPart()
    if not root then return end
    if typeof(target) == "CFrame" then
        root.CFrame = target
    else
        root.CFrame = CFrame.new(target + Vector3.new(0,2,0))
    end
end

local function refreshUnits()
    local data = getPlayerData()
    unitLabels = {}
    labelToKey = {}
    if not data or not data.PlayerUnits then return end
    local sorted = {}
    local seen = {}
    for key, unit in pairs(data.PlayerUnits) do
        local name = unit.PlayerUnitName
        if not name or name == "" then continue end
        local grade = unit.PlayerUnitGrade or unit.Grade or "F"
        local trait = unit.PlayerUnitTrait or unit.Trait or "-"
        local eq = unit.Equipped and " *" or ""
        table.insert(sorted, {key = key, name = name, grade = grade, trait = trait, eq = eq})
    end
    table.sort(sorted, function(a,b)
        if a.name ~= b.name then return a.name < b.name end
        if a.grade ~= b.grade then return a.grade < b.grade end
        if a.trait ~= b.trait then return a.trait < b.trait end
        return a.key < b.key
    end)
    for _, u in ipairs(sorted) do
        local base = u.name .. " [" .. u.grade .. "] (" .. u.trait .. ")" .. u.eq
        local label = base
        local n = 1
        while seen[label] do
            n = n + 1
            label = base .. " #" .. n
        end
        seen[label] = true
        table.insert(unitLabels, label)
        labelToKey[label] = u.key
    end
end

local function returnPos()
    if autoGrade then return POS_GRADE end
    if autoTrait then return POS_TRAIT end
    return getBasePos()
end

local function gradeReached(current, target)
    return (GRADE_ORDER[current] or 0) >= (GRADE_ORDER[target] or 99)
end

local function getSpawnNodes()
    local nodes = CollectionService:GetTagged("VariantTokenSpawn")
    if #nodes == 0 then
        local f = workspace:FindFirstChild("Systems")
        f = f and f:FindFirstChild("VariantTokenSpawns")
        if f then nodes = f:GetChildren() end
    end
    return nodes or {}
end

local function getPlayerPlot()
    local map = workspace:FindFirstChild("Map")
    local plots = map and map:FindFirstChild("PlotModels")
    return plots and plots:FindFirstChild(player.Name)
end

local function findDropperWithUnit(inventoryKey)
    local data = getPlayerData()
    if not data or not data.PlayerUnits then return nil end
    local unit = data.PlayerUnits[inventoryKey]
    if not unit then return nil end
    local plot = getPlayerPlot()
    if not plot then return nil end
    local uName = unit.PlayerUnitName
    local uGrade = unit.PlayerUnitGrade or unit.Grade
    local uTrait = unit.PlayerUnitTrait or unit.Trait
    local uVariant = unit.PlayerUnitVariant or unit.Variant
    local uLevel = unit.PlayerUnitLevel or unit.Level
    for _, part in ipairs(CollectionService:GetTagged("DropperBasePart")) do
        if part:GetAttribute("Type") == "PlayerUnit" and part:IsDescendantOf(plot) then
            if part:GetAttribute("PlayerUnitName") == uName
                and part:GetAttribute("PlayerUnitGrade") == uGrade
                and part:GetAttribute("PlayerUnitTrait") == uTrait
                and part:GetAttribute("PlayerUnitVariant") == uVariant
                and part:GetAttribute("PlayerUnitLevel") == uLevel then
                return part
            end
        end
    end
    return nil
end

local function unequipFromDropper(inventoryKey, statusPara)
    local dropper = findDropperWithUnit(inventoryKey)
    if dropper then
        statusPara:SetDesc("removendo do dropper...")
        removeFromDropper:FireServer(dropper)
        task.wait(0.5)
        return true
    end
    return false
end

collectLoop = function()
    while autoCollect do
        local nodes = getSpawnNodes()
        local root = getRootPart()
        if not root then
            colStatus:SetDesc("sem personagem")
            task.wait(1)
        else
            local got = 0
            for _, node in ipairs(nodes) do
                if not autoCollect then break end
                if node:GetAttribute("Occupied") == true then
                    local v = node:GetAttribute("Variant") or "?"
                    colStatus:SetDesc("coletando: " .. v)
                    collecting = true
                    teleportTo(node.Position)
                    task.wait(0.1)
                    collectRemote:FireServer(node)
                    got = got + 1
                    totalCollected = totalCollected + 1
                    colStatus:SetDesc("coletados: " .. totalCollected)
                    task.wait(LOOP_DELAY)
                end
            end
            if got > 0 then
                teleportTo(returnPos())
            end
            collecting = false
            colStatus:SetDesc(got > 0 and ("+" .. got .. " coletados | total: " .. totalCollected) or ("aguardando... " .. #nodes .. " pontos"))
            task.wait(0.3)
        end
    end
    colStatus:SetDesc("desligado")
end

pcall(function()
    remotes:WaitForChild("VariantTokenSpawned", 5).OnClientEvent:Connect(function(variant)
        if not autoCollect then return end
        colStatus:SetDesc("spawn: " .. tostring(variant))
        task.wait(0.1)
        local root = getRootPart()
        if not root then return end
        collecting = true
        for _, node in ipairs(getSpawnNodes()) do
            if node:GetAttribute("Occupied") == true then
                teleportTo(node.Position)
                task.wait(0.1)
                collectRemote:FireServer(node)
                totalCollected = totalCollected + 1
                colStatus:SetDesc("coletados: " .. totalCollected)
                break
            end
        end
        task.wait(0.15)
        teleportTo(returnPos())
        collecting = false
    end)
end)

gradeLoop = function(gen)
    if not selectedGradeKey then
        grdStatusPara:SetDesc("selecione unidade primeiro")
        autoGrade = false
        Options.AutoGrade:SetValue(false)
        return
    end
    local root = getRootPart()
    if not root then grdStatusPara:SetDesc("sem personagem"); return end
    local target = GRADE_TARGETS[gradeTargetIdx]
    local data = getPlayerData()
    if data and data.PlayerUnits then
        local unit = data.PlayerUnits[selectedGradeKey]
        if unit then
            local cur = unit.PlayerUnitGrade or unit.Grade
            if cur and gradeReached(cur, target) then
                grdStatusPara:SetDesc(cur .. " | ja no alvo!")
                Fluent:Notify({Title = "Grade", Content = cur .. " ja esta no alvo!", Duration = 3})
                autoGrade = false
                Options.AutoGrade:SetValue(false)
                return
            end
        end
    end
    unequipFromDropper(selectedGradeKey, grdStatusPara)
    teleportTo(POS_GRADE)
    task.wait(0.15)
    gradeRolls = 0
    local wasCollecting = false
    while autoGrade and gen == gradeGen do
        if collecting then
            wasCollecting = true
            grdStatusPara:SetDesc("pausado (coletando)")
            task.wait(0.3)
        else
            if wasCollecting then
                wasCollecting = false
                teleportTo(POS_GRADE)
                task.wait(0.15)
            end
            local d = getPlayerData()
            if not d or not d.PlayerUnits then
                grdStatusPara:SetDesc("dados n/a"); break
            end
            local unit = d.PlayerUnits[selectedGradeKey]
            if not unit then grdStatusPara:SetDesc("unidade removida"); break end
            local cur = unit.PlayerUnitGrade or unit.Grade
            if not cur then grdStatusPara:SetDesc("grade nil"); break end
            if gradeReached(cur, target) then
                grdStatusPara:SetDesc(cur .. " | alvo atingido! (" .. gradeRolls .. " rolls)")
                Fluent:Notify({Title = "Grade", Content = cur .. " atingido em " .. gradeRolls .. " rolls!", Duration = 5})
                autoGrade = false
                Options.AutoGrade:SetValue(false)
                break
            end
            gradeRemote:FireServer(selectedGradeKey, false, cur)
            gradeRolls = gradeRolls + 1
            grdStatusPara:SetDesc(cur .. " -> " .. target .. " | " .. gradeRolls .. " rolls")
            task.wait(0.4)
        end
    end
    grdStatusPara:SetDesc("equipando melhores...")
    equipBest:FireServer()
    task.wait(0.3)
    teleportTo(getBasePos())
end

----------------------------------------------------------------
-- background unit refresh (every 10s)
----------------------------------------------------------------
local function findLabelForKey(key)
    for label, k in pairs(labelToKey) do
        if k == key then return label end
    end
    return nil
end

task.spawn(function()
    while task.wait(10) do
        local prevGradeKey = selectedGradeKey
        local prevTraitKey = selectedTraitKey
        refreshUnits()
        pcall(function()
            Options.GradeUnit:SetValues(unitLabels)
            Options.TraitUnit:SetValues(unitLabels)
        end)
        if prevGradeKey then
            local label = findLabelForKey(prevGradeKey)
            if label then
                pcall(function() Options.GradeUnit:SetValue(label) end)
            end
            selectedGradeKey = prevGradeKey
        end
        if prevTraitKey then
            local label = findLabelForKey(prevTraitKey)
            if label then
                pcall(function() Options.TraitUnit:SetValue(label) end)
            end
            selectedTraitKey = prevTraitKey
        end
    end
end)
