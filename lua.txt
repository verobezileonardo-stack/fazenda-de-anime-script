local function parseCashParts(str)
    if not str then return 0, "" end
    str = str:gsub("%$", ""):gsub(",", "."):match("^%s*(.-)%s*$")
    local num, suffix = str:match("^([%d%.]+)%s*(%a*)$")
    return tonumber(num) or 0, (suffix or ""):upper()
end

local function cashGTE(display, threshold)
    local dNum, dSuf = parseCashParts(display)
    local tNum, tSuf = parseCashParts(threshold)
    local dRank = SUFFIX_RANK[dSuf] or -1
    local tRank = SUFFIX_RANK[tSuf] or -1
    if dRank ~= tRank then return dRank > tRank end
    return dNum >= tNum
end

local _rootWarnTick = 0
local function getRootPart()
    local ch = player.Character
    local root = ch and ch:FindFirstChild("HumanoidRootPart")
    if not root and tick() - _rootWarnTick > 10 then
        _rootWarnTick = tick()
        warn("[DEBUG] getRootPart: nil | char=" .. tostring(ch ~= nil))
    end
    return root
end

local function getBasePos()
    if cachedBasePos then return cachedBasePos end
    warn("[DEBUG] getBasePos: calculando pela primeira vez")
    local map = workspace:FindFirstChild("Map")
    local plots = map and map:FindFirstChild("PlotModels")
    local plot = plots and plots:FindFirstChild(player.Name)
    if not plot then warn("[DEBUG] getBasePos: plot nil"); return nil end
    local refPos
    for _, part in ipairs(CollectionService:GetTagged("ConveyorCollectPart")) do
        if part:IsDescendantOf(plot) then refPos = part.Position; break end
    end
    if not refPos then
        for _, desc in ipairs(plot:GetDescendants()) do
            if desc:IsA("BasePart") and (desc.Name:find("Collect") or desc.Name:find("Conveyor")) then
                refPos = desc.Position; break
            end
        end
    end
    if refPos then
        warn("[DEBUG] getBasePos: refPos=" .. tostring(refPos))
        local center = plot:GetPivot().Position
        local dir = Vector3.new(center.X - refPos.X, 0, center.Z - refPos.Z)
        if dir.Magnitude > 0.5 then
            local finalPos = refPos - dir.Unit * 16 + Vector3.new(0, 2, 0)
            local lookAt = finalPos - (refPos - finalPos)
            cachedBasePos = CFrame.new(finalPos, lookAt)
            warn("[DEBUG] getBasePos: cached CFrame ok")
            return cachedBasePos
        end
    else
        warn("[DEBUG] getBasePos: refPos nil, usando pivot")
    end
    cachedBasePos = plot:GetPivot().Position
    return cachedBasePos
end

local function teleportTo(target)
    if not target then warn("[DEBUG] teleportTo: target nil"); return end
    local root = getRootPart()
    if not root then return end
    if typeof(target) == "CFrame" then
        root.CFrame = target
    else
        root.CFrame = CFrame.new(target + Vector3.new(0,2,0))
    end
end

local function buildLabel(u, mode)
    local parts = {u.name}
    if u.variant ~= "Normal" then table.insert(parts, u.variant) end
    if mode == "grade" then
        table.insert(parts, "| " .. (u.grade or "Vazio"))
    else
        table.insert(parts, "| " .. (u.trait or "Vazio"))
    end
    return table.concat(parts, " ") .. u.eq
end

local function refreshUnits()
    local data = getPlayerData()
    gradeLabels, gradeLabelToKey = {}, {}
    traitLabels, traitLabelToKey = {}, {}
    if not data or not data.PlayerUnits then warn("[DEBUG] refreshUnits: sem dados"); return end
    local sorted = {}
    for key, unit in pairs(data.PlayerUnits) do
        local name = unit.PlayerUnitName
        if not name or name == "" then continue end
        table.insert(sorted, {
            key = key, name = name,
            grade = unit.Grade, trait = unit.Trait,
            level = unit.Level or 1, variant = unit.Variant or "Normal",
            eq = unit.Equipped and " [E]" or ""
        })
    end
    table.sort(sorted, function(a,b)
        if a.name ~= b.name then return a.name < b.name end
        if a.variant ~= b.variant then return a.variant < b.variant end
        local ga, gb = GRADE_ORDER[a.grade] or 0, GRADE_ORDER[b.grade] or 0
        if ga ~= gb then return ga < gb end
        local ta, tb = a.trait or "", b.trait or ""
        if ta ~= tb then return ta < tb end
        if a.level ~= b.level then return a.level < b.level end
        return a.key < b.key
    end)
    local seenG, seenT = {}, {}
    for _, u in ipairs(sorted) do
        local gBase = buildLabel(u, "grade")
        local gLabel, gn = gBase, 1
        while seenG[gLabel] do gn = gn + 1; gLabel = gBase .. " #" .. gn end
        seenG[gLabel] = true
        table.insert(gradeLabels, gLabel)
        gradeLabelToKey[gLabel] = u.key

        local tBase = buildLabel(u, "trait")
        local tLabel, tn = tBase, 1
        while seenT[tLabel] do tn = tn + 1; tLabel = tBase .. " #" .. tn end
        seenT[tLabel] = true
        table.insert(traitLabels, tLabel)
        traitLabelToKey[tLabel] = u.key
    end
end

local cachedDroppers, droppersTick = {}, 0
local function getPlotDroppers()
    local now = tick()
    if now - droppersTick < 3 and #cachedDroppers > 0 then return cachedDroppers end
    droppersTick = now
    local plot = getPlayerPlot()
    if not plot then cachedDroppers = {}; return cachedDroppers end
    local out = {}
    for _, part in ipairs(CollectionService:GetTagged("DropperBasePart")) do
        if part:IsDescendantOf(plot) then
            table.insert(out, part)
        end
    end
    cachedDroppers = out
    return out
end

local function findDropperCandidates(inventoryKey)
    warn("[DEBUG] findDropperCandidates: key=" .. tostring(inventoryKey))
    local data = getPlayerData()
    if not data or not data.PlayerUnits then return {}, "sem dados" end
    local unit = data.PlayerUnits[inventoryKey]
    if not unit then return {}, "key invalida" end
    local plot = getPlayerPlot()
    if not plot then return {}, "sem plot" end
    local uName = unit.PlayerUnitName
    local uGrade = unit.Grade and tostring(unit.Grade) or nil
    local uTrait = unit.Trait and tostring(unit.Trait) or nil
    local uLevel = tostring(unit.Level or 1)
    local uVariant = tostring(unit.Variant or "Normal")
    local ranked = {}
    for _, part in ipairs(getPlotDroppers()) do
        if part:GetAttribute("Type") == "PlayerUnit" then
            if part:GetAttribute("PlayerUnitName") == uName then
                local score = 0
                local dGrade = part:GetAttribute("PlayerUnitGrade")
                local dTrait = part:GetAttribute("PlayerUnitTrait")
                local dLevel = tostring(part:GetAttribute("PlayerUnitLevel") or 1)
                local dVariant = tostring(part:GetAttribute("PlayerUnitVariant") or "Normal")
                if dVariant == uVariant then score = score + 4 end
                if dLevel == uLevel then score = score + 1 end
                if uGrade and dGrade and tostring(dGrade) == uGrade then score = score + 3
                elseif not uGrade and not dGrade then score = score + 3 end
                if uTrait and dTrait and tostring(dTrait) == uTrait then score = score + 2
                elseif not uTrait and not dTrait then score = score + 2 end
                local tag = dVariant .. " Lv" .. dLevel .. " " .. (dGrade or "-") .. " " .. (dTrait or "-")
                table.insert(ranked, {part = part, score = score, tag = tag})
            end
        end
    end
    table.sort(ranked, function(a, b) return a.score > b.score end)
    local info = uName .. " " .. uVariant .. " Lv" .. uLevel .. " " .. (uGrade or "-") .. " " .. (uTrait or "-") .. " | " .. #ranked .. " cand"
    return ranked, info
end

local function unequipFromDropper(inventoryKey, statusPara)
    warn("[DEBUG] unequipFromDropper: key=" .. tostring(inventoryKey))
    local ranked, info = findDropperCandidates(inventoryKey)
    if #ranked == 0 then
        statusPara:SetDesc("skip: " .. info)
        task.wait(0.3)
        return false
    end
    for i, cand in ipairs(ranked) do
        statusPara:SetDesc("remove #" .. i .. "/" .. #ranked .. " s=" .. cand.score .. "/10 " .. cand.tag)
        warn("[DEBUG] unequip: removeFromDropper #" .. i .. " score=" .. cand.score)
        removeFromDropper:FireServer(cand.part)
        task.wait(0.5)
        local d = getPlayerData()
        if d and d.PlayerUnits then
            local u = d.PlayerUnits[inventoryKey]
            if u and not u.Equipped then
                statusPara:SetDesc("ok: removido em " .. i .. " tentativa(s)")
                return true
            end
        end
    end
    statusPara:SetDesc("falhou: " .. #ranked .. " tentativas, ainda equipado")
    return false
end

collectLoop = function()
    warn("[DEBUG] collectLoop: iniciado")
    while autoCollect do
        local root = getRootPart()
        if not root then
            colStatus:SetDesc("sem personagem")
            task.wait(1)
        elseif crateBusy then
            colStatus:SetDesc("aguardando (crate busy)")
            task.wait(1)
        else
            local got = 0
            for _, node in ipairs(getSpawnNodes()) do
                if not autoCollect or crateBusy then break end
                if node:GetAttribute("Occupied") == true then
                    collecting = true
                    colStatus:SetDesc("coletando: " .. (node:GetAttribute("Variant") or "?"))
                    warn("[DEBUG] collectLoop: variant=" .. tostring(node:GetAttribute("Variant")) .. " pos=" .. tostring(node.Position))
                    teleportTo(node.Position)
                    task.wait(0.1)
                    collectRemote:FireServer(node)
                    got = got + 1
                    totalCollected = totalCollected + 1
                    task.wait(LOOP_DELAY)
                end
            end
            if got > 0 then
                colStatus:SetDesc("+" .. got .. " | total: " .. totalCollected)
                teleportTo(returnPos())
            else
                colStatus:SetDesc("aguardando...")
            end
            collecting = false
            task.wait(0.5)
        end
    end
    colStatus:SetDesc("desligado")
end

gradeLoop = function(gen)
    warn("[DEBUG] gradeLoop: gen=" .. tostring(gen) .. " key=" .. tostring(selectedGradeKey))
    if not selectedGradeKey then
        grdStatusPara:SetDesc("selecione unidade primeiro")
        autoGrade = false
        Options.AutoGrade:SetValue(false)
        return
    end
    local root = getRootPart()
    if not root then grdStatusPara:SetDesc("sem personagem"); return end
    local target = GRADE_TARGETS[gradeTargetIdx]
    local data = getPlayerData()
    if not data or not data.PlayerUnits or not data.PlayerUnits[selectedGradeKey] then
        grdStatusPara:SetDesc("unidade nao encontrada"); autoGrade = false; Options.AutoGrade:SetValue(false); return
    end
    local unitInfo = data.PlayerUnits[selectedGradeKey]
    local unitName = unitInfo.PlayerUnitName or "?"
    local cur = unitInfo.Grade
    grdStatusPara:SetDesc(unitName .. " [" .. tostring(cur) .. "] -> " .. target)
    if cur and gradeReached(cur, target) then
        grdStatusPara:SetDesc(unitName .. " [" .. tostring(cur) .. "] ja no alvo!")
        Fluent:Notify({Title = "Grade", Content = unitName .. " " .. tostring(cur) .. " ja esta no alvo!", Duration = 3})
        autoGrade = false
        Options.AutoGrade:SetValue(false)
        return
    end
    if unitInfo.Equipped then
        unequipFromDropper(selectedGradeKey, grdStatusPara)
    end
    teleportTo(POS_GRADE)
    task.wait(0.15)
    gradeRolls = 0
    local reachedTarget = false
    local wasCollecting = false
    while autoGrade and gen == gradeGen do
        if collecting or crateBusy then
            wasCollecting = true
            grdStatusPara:SetDesc("pausado (" .. (crateBusy and "crate" or "coletando") .. ")")
            task.wait(0.3)
        else
            if wasCollecting then
                wasCollecting = false
                teleportTo(POS_GRADE)
                task.wait(0.15)
            end
            local d = getPlayerData()
            if not d or not d.PlayerUnits then
                grdStatusPara:SetDesc("dados n/a"); break
            end
            local unit = d.PlayerUnits[selectedGradeKey]
            if not unit then grdStatusPara:SetDesc("unidade removida"); break end
            local cur = unit.Grade
            if not cur then grdStatusPara:SetDesc("grade nil"); break end
            if gradeReached(cur, target) then
                grdStatusPara:SetDesc(unitName .. " " .. cur .. " | alvo atingido! (" .. gradeRolls .. " rolls)")
                Fluent:Notify({Title = "Grade", Content = unitName .. " " .. cur .. " atingido em " .. gradeRolls .. " rolls!", Duration = 5})
                reachedTarget = true
                autoGrade = false
                Options.AutoGrade:SetValue(false)
                break
            end
            if gradeRolls % 25 == 0 then warn("[DEBUG] gradeLoop: roll=" .. gradeRolls .. " cur=" .. tostring(cur)) end
            gradeRemote:FireServer(selectedGradeKey, false, cur)
            gradeRolls = gradeRolls + 1
            grdStatusPara:SetDesc(unitName .. " " .. cur .. " -> " .. target .. " | " .. gradeRolls)
            task.wait(0.4)
        end
    end
    if reachedTarget then
        grdStatusPara:SetDesc("equipando melhores...")
        equipBest:FireServer()
        task.wait(0.3)
    end
    teleportTo(getBasePos())
end

crateLoop = function(gen)
    warn("[DEBUG] crateLoop: iniciado gen=" .. tostring(gen))
    local plot = getPlayerPlot()
    if not plot then crateStatus:SetDesc("sem plot"); autoCrate = false; Options.AutoCrate:SetValue(false); return end

    if not pickupCrateRemote or not depositCrateRemote or not collectPolishedRemote or not sellCrateRemote then
        crateStatus:SetDesc("remotes crate ausentes!")
        warn("[DEBUG] crateLoop: remotes nil, abortando")
        autoCrate = false; Options.AutoCrate:SetValue(false); return
    end

    local conveyorPart
    for _, part in ipairs(CollectionService:GetTagged("ConveyorCollectPart")) do
        if part:IsDescendantOf(plot) then conveyorPart = part; break end
    end
    if not conveyorPart then crateStatus:SetDesc("ConveyorCollectPart nao encontrado"); autoCrate = false; Options.AutoCrate:SetValue(false); return end

    local polisherCollectPart
    for _, model in ipairs(CollectionService:GetTagged("CratePolisherModel")) do
        if model:IsDescendantOf(plot) then
            local phys = model:FindFirstChild("PhysicalModel")
            local cp = phys and phys:FindFirstChild("CollectPart")
            if cp then polisherCollectPart = cp; break end
        end
    end
    if not polisherCollectPart then crateStatus:SetDesc("Polisher nao encontrado"); autoCrate = false; Options.AutoCrate:SetValue(false); return end

    warn("[DEBUG] crateLoop: conveyor=" .. conveyorPart:GetFullName())
    warn("[DEBUG] crateLoop: polisher=" .. polisherCollectPart:GetFullName())

    local sellNPC
    for _, part in ipairs(CollectionService:GetTagged("SellNPC")) do
        sellNPC = part; break
    end
    if sellNPC then warn("[DEBUG] crateLoop: sellNPC=" .. sellNPC:GetFullName()) end

    while autoCrate and gen == crateGen do
        local display = conveyorPart:GetAttribute("PendingCashDisplay") or "0"
        local balls = conveyorPart:GetAttribute("PendingBallCount") or 0
        crateStatus:SetDesc("$" .. display .. " (" .. balls .. " balls) | alvo: " .. crateThreshold)

        if balls <= 0 or not cashGTE(display, crateThreshold) then
            task.wait(5)
            continue
        end

        -- esperar collectLoop terminar antes de iniciar ciclo
        while collecting do task.wait(0.3) end
        if not autoCrate or gen ~= crateGen then break end

        warn("[DEBUG] crateLoop: THRESHOLD! display=" .. display .. " balls=" .. balls)
        crateBusy = true
        local cycleOk, cycleErr = pcall(function()

            -- fase 1: pickup
            crateStatus:SetDesc("fase 1/6: coletando crate...")
            local ballsBefore = conveyorPart:GetAttribute("PendingBallCount") or 0
            teleportTo(conveyorPart.Position)
            task.wait(0.3)
            pickupCrateRemote:FireServer(conveyorPart)
            task.wait(0.8)
            local ballsAfter = conveyorPart:GetAttribute("PendingBallCount") or 0
            local char = player.Character
            local hasTool = false
            if char then
                for _, t in ipairs(char:GetChildren()) do
                    if t:IsA("Tool") and t:GetAttribute("InventoryType") == "Crate" then hasTool = true; break end
                end
            end
            warn("[DEBUG] crateLoop: pickup balls " .. ballsBefore .. "->" .. ballsAfter .. " tool=" .. tostring(hasTool))
            if not hasTool then
                crateStatus:SetDesc("pickup falhou — sem crate na mao")
                warn("[DEBUG] crateLoop: pickup FALHOU, abortando ciclo")
                return false
            end

            -- fase 2: deposit no polisher
            crateStatus:SetDesc("fase 2/6: depositando no polisher...")
            teleportTo(polisherCollectPart.Position)
            task.wait(0.3)
            depositCrateRemote:FireServer(polisherCollectPart)
            task.wait(0.5)
            local data = getPlayerData()
            local pendingCash = data and data.Polisher and data.Polisher.PendingCash or 0
            warn("[DEBUG] crateLoop: deposit done, Polisher.PendingCash=" .. tostring(pendingCash))

            -- fase 3: aguardar polish
            crateStatus:SetDesc("fase 3/6: aguardando polish...")
            local polishStart = tick()
            local polishTimeout = 300
            local polishOk = false
            while autoCrate and gen == crateGen do
                local polished = polisherCollectPart:GetAttribute("PolishedBallCount") or 0
                local d2 = getPlayerData()
                local pc = d2 and d2.Polisher and d2.Polisher.PendingCash or 0
                if polished > 0 and pc == 0 then
                    warn("[DEBUG] crateLoop: polish OK em " .. string.format("%.1fs", tick() - polishStart) .. " polished=" .. polished)
                    polishOk = true
                    break
                end
                local elapsed = tick() - polishStart
                if elapsed > polishTimeout then
                    warn("[DEBUG] crateLoop: polish TIMEOUT " .. polishTimeout .. "s, polished=" .. polished .. " pending=" .. tostring(pc))
                    crateStatus:SetDesc("polish timeout!")
                    break
                end
                crateStatus:SetDesc("fase 3/6: polindo... " .. string.format("%.0fs", polishTimeout - elapsed) .. " | pending=$" .. tostring(pc))
                task.wait(3)
            end
            if not polishOk then return false end

            -- fase 4: cash potion
            if useCashPotion and useItemRemote then
                local potionsFolder = player:FindFirstChild("Potions")
                local active = potionsFolder and (potionsFolder:GetAttribute("CashPotion") or 0) > 0
                if not active then
                    crateStatus:SetDesc("fase 4/6: usando Cash Potion...")
                    useItemRemote:FireServer("CashPotion")
                    warn("[DEBUG] crateLoop: UseItem CashPotion fired")
                    task.wait(1)
                else
                    warn("[DEBUG] crateLoop: CashPotion ja ativa, skip")
                end
            end

            -- fase 5: collect polished
            crateStatus:SetDesc("fase 5/6: coletando crate polida...")
            local polishedBefore = polisherCollectPart:GetAttribute("PolishedBallCount") or 0
            collectPolishedRemote:FireServer(polisherCollectPart)
            task.wait(0.8)
            local polishedAfter = polisherCollectPart:GetAttribute("PolishedBallCount") or 0
            warn("[DEBUG] crateLoop: collect polished=" .. polishedBefore .. "->" .. polishedAfter)

            -- fase 6: sell
            crateStatus:SetDesc("fase 6/6: vendendo...")
            if sellNPC then
                local sellPos = sellNPC:IsA("BasePart") and sellNPC.Position or sellNPC:GetPivot().Position
                teleportTo(sellPos)
                task.wait(0.3)
            end
            local sold = 0
            char = player.Character
            if char then
                for _, tool in ipairs(char:GetChildren()) do
                    if tool:IsA("Tool") and tool:GetAttribute("InventoryType") == "Crate" then
                        local guid = tool:GetAttribute("InventoryKey")
                        if guid then
                            sellCrateRemote:FireServer(guid)
                            sold = sold + 1
                            task.wait(0.3)
                        end
                    end
                end
            end
            warn("[DEBUG] crateLoop: vendidas=" .. sold)
            return true
        end)

        crateBusy = false
        if not cycleOk then
            warn("[DEBUG] crateLoop: ERRO no ciclo: " .. tostring(cycleErr))
            crateStatus:SetDesc("erro: " .. tostring(cycleErr))
        elseif cycleErr == false then
            crateStatus:SetDesc("ciclo abortado, retentando em 10s...")
            task.wait(10)
        else
            crateStatus:SetDesc("ciclo completo! aguardando acumular...")
        end
        teleportTo(returnPos())
        task.wait(3)
    end
    crateBusy = false
    crateStatus:SetDesc("desligado")
end

local function findLabelForKey(map, key)
    for label, k in pairs(map) do
        if k == key then return label end
    end
    return nil
end

        local prevGradeKey = selectedGradeKey
        local prevTraitKey = selectedTraitKey
        refreshUnits()

            Options.GradeUnit:SetValues(gradeLabels)
            Options.TraitUnit:SetValues(traitLabels)
        end)
        if prevGradeKey then
            local label = findLabelForKey(gradeLabelToKey, prevGradeKey)

            selectedGradeKey = prevGradeKey
        end
        if prevTraitKey then

            if label then pcall(function() Options.TraitUnit:SetValue(label) end) end
            selectedTraitKey = prevTraitKey
        end
    end
end)

-- anti-afk (simula input real — Roblox core checa UserInputService)

local function fakeInput()
    local ok = pcall(function()
        if keypress and keyrelease then
            keypress(0x20)
            task.wait(0.05)
            keyrelease(0x20)
            return
        end
    end)
    if ok then return end
    pcall(function()
        local vim = game:GetService("VirtualInputManager")
        vim:SendKeyEvent(true, Enum.KeyCode.Space, false, game)
        task.wait(0.05)
        vim:SendKeyEvent(false, Enum.KeyCode.Space, false, game)
    end)
    pcall(function()
        local vu = game:GetService("VirtualUser")
        vu:CaptureController()
        vu:ClickButton2(Vector2.new())
    end)
end

player.Idled:Connect(fakeInput)

        fakeInput()
    end
end)
